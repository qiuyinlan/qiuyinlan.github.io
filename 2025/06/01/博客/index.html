这个 Token 可以用来做什么？

具体能干嘛，取决于你在创建 token 时勾选了哪些权限。常见用途如下：

1. Hexo 或其他博客自动部署

  • 把本地生成的静态页面自动上传到 GitHub Pages 仓库。

  • 你在 hexo deploy 的配置里写入这个 token，就可以用命令一键部署。

2. GitHub Actions 等自动化脚本

  • 自动运行 CI/CD（持续集成/部署）流程。

  • 推送代码、发布 Release、更新文件等操作无需手动登录。

3. 命令行操作 GitHub

  • 用 curl、gh 工具、Git 客户端等命令行工具执行 GitHub API 操作。

  • 适用于需要自动化控制 GitHub 仓库的场景。

4. 访问私有仓库

  • 如果你有私有仓库，token 允许你访问或修改这些仓库。

import os  
import shutil  
import re  
import subprocess  
from datetime import datetime  
  
# 你的配置  
OBSIDIAN_DIR = "/home/xuzichun/文档/Obsidian Vault"  
HEXO_POSTS_DIR = "/home/xuzichun/文档/Obsidian Vault/myblog/source/_posts"  
HEXO_IMG_DIR = "/home/xuzichun/文档/Obsidian Vault/myblog/source/images"  
BLOG_ROOT = "/home/xuzichun/文档/Obsidian Vault/myblog"  
PUBLISH_TAG = ""  
PICTURES_DIR = "/home/xuzichun/文档/Obsidian Vault/pictures"  # 你的图片统一存放目录  
  
def ensure_dir(path):  
    if not os.path.exists(path):  
        os.makedirs(path)  
  
def copy_image(src_path, dest_path):  
    ensure_dir(os.path.dirname(dest_path))  
    shutil.copy2(src_path, dest_path)  
  
def extract_tags(content):  
    tags = set(re.findall(r'#(\w+)', content))  
    if 'publish' in tags:  
        tags.remove('publish')  
    return list(tags)  
  
def create_front_matter(title, date_str, tags):  
    fm = "---\n"  
    fm += f"title: {title}\n"  
    fm += f"date: {date_str}\n"  
    if tags:  
        fm += "tags:\n"  
        for tag in tags:  
            fm += f"- {tag}\n"  
    fm += "categories:\n- blog\n"  
    fm += "---\n\n"  
    return fm  
  
def process_markdown(md_path):  
    with open(md_path, "r", encoding="utf-8") as f:  
        content = f.read()  
  
    if PUBLISH_TAG not in content:  
        return False  
  
    img_paths = re.findall(r'!\[\[([^\]]+)\]\]', content)  # 找 ![[图片名]] 格式  
  
    for img_name in img_paths:  
        # 替换图片名中的空格为下划线  
        safe_img_name = img_name.replace(" ", "_")  
  
        abs_img_path = os.path.join(PICTURES_DIR, img_name)  # 从 pictures 目录取图片  
        if not os.path.exists(abs_img_path):  
            print(f"图片不存在，跳过: {abs_img_path}")  
            continue  
        dest_img_path = os.path.join(HEXO_IMG_DIR, safe_img_name)  
        copy_image(abs_img_path, dest_img_path)  
  
        # 替换 ![[xxx.png]] 为 ![](/images/xxx.png)，用安全名  
        content = content.replace(f"![[{img_name}]]", f"![](/images/{safe_img_name})")  
  
    title = os.path.splitext(os.path.basename(md_path))[0]  
    mtime = os.path.getmtime(md_path)  
    date_str = datetime.fromtimestamp(mtime).strftime("%Y-%m-%d %H:%M:%S")  
    tags = extract_tags(content)  
  
    front_matter = create_front_matter(title, date_str, tags)  
  
    content = re.sub(r'#\w+', '', content)  # 去除所有标签  
  
    dest_md_path = os.path.join(HEXO_POSTS_DIR, os.path.basename(md_path))  
    with open(dest_md_path, "w", encoding="utf-8") as f:  
        f.write(front_matter + content.strip() + "\n")  
  
    print(f"同步完成: {md_path} -> {dest_md_path}")  
    return True  
  
def sync_notes():  
    any_synced = False  
    for root, dirs, files in os.walk(OBSIDIAN_DIR):  
        for file in files:  
            if file.endswith(".md"):  
                md_path = os.path.join(root, file)  
                synced = process_markdown(md_path)  
                if synced:  
                    any_synced = True  
    return any_synced  
  
def clean_removed_posts():  
    """  
    删除 Hexo _posts 目录中  
    但 Obsidian 中已无对应  标记的 md 文件，  
    保持同步删除已取消发布的笔记。  
    """  
    # 收集 Obsidian 中所有带  的 md 文件名  
    published_md_names = set()  
    for root, dirs, files in os.walk(OBSIDIAN_DIR):  
        for file in files:  
            if file.endswith(".md"):  
                md_path = os.path.join(root, file)  
                with open(md_path, encoding="utf-8") as f:  
                    content = f.read()  
                if PUBLISH_TAG in content:  
                    published_md_names.add(file)  
  
    # 遍历 Hexo _posts 目录，删除无对应发布文件的 md 文件  
    for file in os.listdir(HEXO_POSTS_DIR):  
        if file.endswith(".md") and file not in published_md_names:  
            path_to_delete = os.path.join(HEXO_POSTS_DIR, file)  
            print(f"删除 Hexo 中无对应笔记的文件: {path_to_delete}")  
            os.remove(path_to_delete)  
  
def run_cmd(cmd, cwd=None):  
    result = subprocess.run(cmd, shell=True, cwd=cwd,  
                            stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)  
    if result.returncode != 0:  
        print(f"命令失败: {cmd}\n错误: {result.stderr}")  
        return False  
    else:  
        print(result.stdout)  
        return True  
  
def main():  
    ensure_dir(HEXO_POSTS_DIR)  
    ensure_dir(HEXO_IMG_DIR)  
  
    changed = sync_notes()  
    clean_removed_posts()  # 新增：同步删除无对应笔记的 Hexo md 文件  
  
    if not changed:  
        print("没有新的内容需要同步，退出。")  
        return  
  
    print("开始生成静态文件...")  
    if not run_cmd("hexo clean", BLOG_ROOT):  
        return  
    if not run_cmd("hexo generate", BLOG_ROOT):  
        return  
  
    print("开始 git 提交并推送...")  
    run_cmd("git add .", BLOG_ROOT)  
    commit_msg = f"自动同步博客更新: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"  
    if not run_cmd(f'git commit -m "{commit_msg}"', BLOG_ROOT):  
        print("没有变化可提交，跳过 commit。")  
    else:  
        run_cmd("git push", BLOG_ROOT)  
  
    print("部署博客...")  
    run_cmd("hexo deploy", BLOG_ROOT)  
  
    print("全部操作完成。")  
  
if __name__ == "__main__":  
    main()

import os  
import shutil  
import re  
from datetime import datetime  
  
# 请根据你实际路径确认  
OBSIDIAN_DIR = "/home/xuzichun/文档/Obsidian Vault/git_publish"   # Obsidian笔记文件夹  
HEXO_POSTS_DIR = "/home/xuzichun/文档/Obsidian Vault/myblog/source/_posts"  # Hexo文章文件夹  
HEXO_IMG_DIR = "/home/xuzichun/文档/Obsidian Vault/myblog/source/images"   # Hexo图片文件夹  
PUBLISH_TAG = ""  
  
def ensure_dir(path):  
    if not os.path.exists(path):  
        os.makedirs(path)  
  
def copy_image(src_path, dest_path):  
    ensure_dir(os.path.dirname(dest_path))  
    # 复制时加时间戳防止重名覆盖  
    if os.path.exists(dest_path):  
        base, ext = os.path.splitext(dest_path)  
        timestamp = datetime.now().strftime("%Y%m%d%H%M%S")  
        dest_path = f"{base}_{timestamp}{ext}"  
    shutil.copy2(src_path, dest_path)  
    return dest_path  
  
def extract_tags(content):  
    tags = set(re.findall(r'#(\w+)', content))  
    if 'publish' in tags:  
        tags.remove('publish')  
    return list(tags)  
  
def create_front_matter(title, date_str, tags):  
    fm = "---\n"  
    fm += f"title: {title}\n"  
    fm += f"date: {date_str}\n"  
    if tags:  
        fm += "tags:\n"  
        for tag in tags:  
            fm += f"- {tag}\n"  
    fm += "categories:\n- blog\n"  
    fm += "---\n\n"  
    return fm  
  
def process_markdown(md_path):  
    with open(md_path, "r", encoding="utf-8") as f:  
        content = f.read()  
  
    if PUBLISH_TAG not in content:  
        # 没有publish标签跳过  
        return False  
  
    # 找出标准Markdown格式图片路径 ![alt](path)  
    md_img_paths = re.findall(r'!\[.*?\]\((.*?)\)', content)  
    # 找出Obsidian内部格式图片路径 ![[image.png]]  
    obsidian_img_paths = re.findall(r'!\[\[(.*?)\]\]', content)  
  
    # 处理所有图片路径，复制图片并替换路径  
    for img_path in md_img_paths + obsidian_img_paths:  
        # 跳过网络图片  
        if img_path.startswith("http://") or img_path.startswith("https://"):  
            continue  
  
        # 处理绝对路径或相对路径  
        if os.path.isabs(img_path):  
            abs_img_path = img_path  
        else:  
            abs_img_path = os.path.normpath(os.path.join(os.path.dirname(md_path), img_path))  
  
        if not os.path.exists(abs_img_path):  
            print(f"图片不存在，跳过: {abs_img_path}")  
            continue  
  
        img_name = os.path.basename(abs_img_path)  
        dest_img_path = os.path.join(HEXO_IMG_DIR, img_name)  
        dest_img_path = copy_image(abs_img_path, dest_img_path)  
  
        # 取新文件名替换路径，改为 /images/xxxx.png  
        new_img_name = os.path.basename(dest_img_path)  
        new_path = f"/images/{new_img_name}"  
  
        # 替换两种格式  
        if img_path in md_img_paths:  
            # 替换标准Markdown路径  
            content = content.replace(f"({img_path})", f"({new_path})")  
        else:  
            # 替换Obsidian内链格式 ![[img]] -> ![](/images/img)  
            content = content.replace(f"![[{img_path}]]", f"![]({new_path})")  
  
    # 生成Front-matter  
    title = os.path.splitext(os.path.basename(md_path))[0]  
    mtime = os.path.getmtime(md_path)  
    date_str = datetime.fromtimestamp(mtime).strftime("%Y-%m-%d %H:%M:%S")  
    tags = extract_tags(content)  
  
    front_matter = create_front_matter(title, date_str, tags)  
  
    # 去除正文中所有标签（包括 ）  
    content = re.sub(r'#\w+', '', content)  
  
    dest_md_path = os.path.join(HEXO_POSTS_DIR, os.path.basename(md_path))  
    with open(dest_md_path, "w", encoding="utf-8") as f:  
        f.write(front_matter + content.strip() + "\n")  
  
    print(f"同步完成: {md_path} -> {dest_md_path}")  
    return True  
  
def sync_notes():  
    ensure_dir(HEXO_POSTS_DIR)  
    ensure_dir(HEXO_IMG_DIR)  
  
    for root, dirs, files in os.walk(OBSIDIAN_DIR):  
        for file in files:  
            if file.endswith(".md"):  
                md_path = os.path.join(root, file)  
                process_markdown(md_path)  
  
if __name__ == "__main__":  
    sync_notes()  
    print("所有笔记同步完成！")

博客知识

  • hexo server（或简写 hexo s）是Hexo 自带的本地预览命令 。

  • 它会启动一个本地的 Web 服务器（默认是//localhost:4000），把你的博客内容渲染成网页，方便你在本地电脑上预览 博客效果。

  • 这个预览的内容是你本地的博客文件生成的“静态网页”，并没有发布到线上网站 。

博客更新流程

  1. 编辑文章和内容 （在 Obsidian 或 Hexo 目录）

  2. 运行脚本把内容同步到 Hexo 博客项目 （同步 Markdown + 图片）

  3. 运行hexo server 本地预览，确认没问题

  4. 运行hexo generate（简写 hexo g）生成静态网页文件

  5. 运行hexo deploy（简写 hexo d）把生成好的网页上传到 GitHub Pages 或你的远程服务器

    • 也可以通过 git push 把 Hexo 项目推送到 GitHub，然后 GitHub Actions 自动帮你部署。

import os  
import shutil  
import re  
import subprocess  
from datetime import datetime  
  
# 你的配置  
OBSIDIAN_DIR = "/home/xuzichun/文档/Obsidian Vault"  
HEXO_POSTS_DIR = "/home/xuzichun/文档/Obsidian Vault/myblog/source/_posts"  
HEXO_IMG_DIR = "/home/xuzichun/文档/Obsidian Vault/myblog/source/images"  
BLOG_ROOT = "/home/xuzichun/文档/Obsidian Vault/myblog"  
PUBLISH_TAG = ""  
PICTURES_DIR = "/home/xuzichun/文档/Obsidian Vault/pictures"  # 你的图片统一存放目录  
  
def ensure_dir(path):  
    if not os.path.exists(path):  
        os.makedirs(path)  
  
def copy_image(src_path, dest_path):  
    ensure_dir(os.path.dirname(dest_path))  
    shutil.copy2(src_path, dest_path)  
  
def extract_tags(content):  
    tags = set(re.findall(r'#(\w+)', content))  
    if 'publish' in tags:  
        tags.remove('publish')  
    return list(tags)  
  
def create_front_matter(title, date_str, tags):  
    fm = "---\n"  
    fm += f"title: {title}\n"  
    fm += f"date: {date_str}\n"  
    if tags:  
        fm += "tags:\n"  
        for tag in tags:  
            fm += f"- {tag}\n"  
    fm += "categories:\n- blog\n"  
    fm += "---\n\n"  
    return fm  
  
def process_markdown(md_path):  
    with open(md_path, "r", encoding="utf-8") as f:  
        content = f.read()  
  
    if PUBLISH_TAG not in content:  
        return False  
  
    img_paths = re.findall(r'!\[\[([^\]]+)\]\]', content)  # 找 ![[图片名]] 格式  
  
    for img_name in img_paths:  
        abs_img_path = os.path.join(PICTURES_DIR, img_name)  # 从 pictures 目录取图片  
        if not os.path.exists(abs_img_path):  
            print(f"图片不存在，跳过: {abs_img_path}")  
            continue  
        dest_img_path = os.path.join(HEXO_IMG_DIR, img_name)  
        copy_image(abs_img_path, dest_img_path)  
  
        # 替换 ![[xxx.png]] 为 ![](/images/xxx.png)  
        content = content.replace(f"![[{img_name}]]", f"![](/images/{img_name})")  
  
    title = os.path.splitext(os.path.basename(md_path))[0]  
    mtime = os.path.getmtime(md_path)  
    date_str = datetime.fromtimestamp(mtime).strftime("%Y-%m-%d %H:%M:%S")  
    tags = extract_tags(content)  
  
    front_matter = create_front_matter(title, date_str, tags)  
  
    content = re.sub(r'#\w+', '', content)  # 去除所有标签  
  
    dest_md_path = os.path.join(HEXO_POSTS_DIR, os.path.basename(md_path))  
    with open(dest_md_path, "w", encoding="utf-8") as f:  
        f.write(front_matter + content.strip() + "\n")  
  
    print(f"同步完成: {md_path} -> {dest_md_path}")  
    return True  
  
def sync_notes():  
    any_synced = False  
    for root, dirs, files in os.walk(OBSIDIAN_DIR):  
        for file in files:  
            if file.endswith(".md"):  
                md_path = os.path.join(root, file)  
                synced = process_markdown(md_path)  
                if synced:  
                    any_synced = True  
    return any_synced  
  
def run_cmd(cmd, cwd=None):  
    result = subprocess.run(cmd, shell=True, cwd=cwd,  
                            stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)  
    if result.returncode != 0:  
        print(f"命令失败: {cmd}\n错误: {result.stderr}")  
        return False  
    else:  
        print(result.stdout)  
        return True  
  
def main():  
    ensure_dir(HEXO_POSTS_DIR)  
    ensure_dir(HEXO_IMG_DIR)  
  
    changed = sync_notes()  
    if not changed:  
        print("没有新的内容需要同步，退出。")  
        return  
  
    print("开始生成静态文件...")  
    if not run_cmd("hexo clean", BLOG_ROOT):  
        return  
    if not run_cmd("hexo generate", BLOG_ROOT):  
        return  
  
    print("开始 git 提交并推送...")  
    run_cmd("git add .", BLOG_ROOT)  
    commit_msg = f"自动同步博客更新: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"  
    if not run_cmd(f'git commit -m "{commit_msg}"', BLOG_ROOT):  
        print("没有变化可提交，跳过 commit。")  
    else:  
        run_cmd("git push", BLOG_ROOT)  
  
    print("部署博客...")  
    run_cmd("hexo deploy", BLOG_ROOT)  
  
    print("全部操作完成。")  
  
if __name__ == "__main__":  
    main()
